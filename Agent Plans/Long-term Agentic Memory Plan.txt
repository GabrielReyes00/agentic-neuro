PHASE 1: RECIPROCAL RANK FUSION FOR recall_episodes()
================================================================================

STATUS: COMPLETE

GOAL: Replace the current naive set-union merge in recall_episodes() with
principled RRF scoring that fuses three retrieval streams: structured SQLite
filters, keyword matching, and LanceDB semantic search. This produces
significantly more relevant episodic recall results.

BACKGROUND: The current recall_episodes() method in knowledge_graph.py (line
~4080) does structured SQLite filtering + keyword LIKE matching, then merges
by set union with structured matches first. There is no scoring — items from
the structured stream always come before keyword-only matches. Meanwhile,
lance_retriever.py has search_episodic_memory() (line ~3366) for semantic
search, but it's completely disconnected from recall_episodes(). The two
systems never talk to each other.

RRF (Reciprocal Rank Fusion) is a principled fusion method:
  RRF_score(item) = sum over streams: 1 / (k + rank_in_stream)
where k=60 is standard. Items appearing in multiple streams get boosted.
Items only in one stream still contribute. No normalization needed.

4. REPLACE the current merge logic (lines ~4164-4189) with RRF fusion:

   ```python
   # ── RRF Fusion across three streams ──
   K = 60  # standard RRF constant

   # Build per-stream ranked lists (exchange_id -> rank, 0-indexed)
   struct_ranking: dict[int, int] = {}
   for rank, row in enumerate(struct_rows):
       xid = row["exchange_id"]
       struct_ranking[xid] = rank

   keyword_ranking: dict[int, int] = {}
   if keyword_ids:
       # Keyword results don't have a natural order — rank by recency
       # Fetch them with ordering
       kw_ordered_ids = list(keyword_ids)
       if kw_ordered_ids:
           placeholders = ",".join("?" for _ in kw_ordered_ids)
           kw_rows = self.conn.execute(
               f"SELECT exchange_id FROM learning_exchanges "
               f"WHERE exchange_id IN ({placeholders}) "
               f"ORDER BY session_ts DESC, turn_number DESC",
               kw_ordered_ids,
           ).fetchall()
           for rank, row in enumerate(kw_rows):
               keyword_ranking[row["exchange_id"]] = rank

   semantic_ranking: dict[int, int] = {}
   for rank, sr in enumerate(semantic_rows):
       xid = sr.get("exchange_id", -1)
       if xid > 0:
           semantic_ranking[xid] = rank

   # Collect all candidate exchange_ids
   all_candidate_ids = set(struct_ranking) | set(keyword_ranking) | set(semantic_ranking)

   # Compute RRF score for each candidate
   rrf_scores: dict[int, float] = {}
   for xid in all_candidate_ids:
       score = 0.0
       if xid in struct_ranking:
           score += 1.0 / (K + struct_ranking[xid])
       if xid in keyword_ranking:
           score += 1.0 / (K + keyword_ranking[xid])
       if xid in semantic_ranking:
           score += 1.0 / (K + semantic_ranking[xid])
       rrf_scores[xid] = score

   # Sort by RRF score descending, take top max_results
   sorted_ids = sorted(rrf_scores.keys(), key=lambda x: -rrf_scores[x])[:max_results]

   # Fetch full rows for the selected IDs
   # (reuse already-fetched struct_rows where possible to avoid extra DB hits)
   struct_by_id = {r["exchange_id"]: dict(r) for r in struct_rows}
   exchanges = []
   missing_ids = []
   for xid in sorted_ids:
       if xid in struct_by_id:
           exchanges.append(struct_by_id[xid])
       else:
           missing_ids.append(xid)

   if missing_ids:
       placeholders = ",".join("?" for _ in missing_ids)
       extra = self.conn.execute(
           f"SELECT le.*, t.display_name AS topic_display "
           f"FROM learning_exchanges le "
           f"LEFT JOIN topics t ON le.topic_id = t.topic_id "
           f"WHERE le.exchange_id IN ({placeholders})",
           missing_ids,
       ).fetchall()
       extra_by_id = {r["exchange_id"]: dict(r) for r in extra}
       for xid in missing_ids:
           if xid in extra_by_id:
               exchanges.append(extra_by_id[xid])

   # Re-sort exchanges by RRF score order
   id_order = {xid: i for i, xid in enumerate(sorted_ids)}
   exchanges.sort(key=lambda e: id_order.get(e["exchange_id"], 999))
   ```

   ```python
   def recall_episodes_compact(
       self,
       query: str = "",
       topic_name: str | None = None,
       domain: str | None = None,
       error_type: str | None = None,
       answer_correct: int | None = None,
       skill: str | None = None,
       days_back: int = 90,
       max_results: int = 10,
   ) -> dict:
       """Compact version of recall_episodes() for preflight token savings.

       Returns the same structure but with truncated text fields and a
       one-line summary per exchange instead of full verbatim content.
       Includes exchange_ids so full content can be fetched on demand
       via exchange_history().
       """
       full = self.recall_episodes(
           query=query, topic_name=topic_name, domain=domain,
           error_type=error_type, answer_correct=answer_correct,
           skill=skill, days_back=days_back, max_results=max_results,
       )

       compact_exchanges = []
       for ex in full.get("exchanges", []):
           correct_label = {0: "INCORRECT", 1: "PARTIAL", 2: "CORRECT"}.get(
               ex.get("answer_correct", -1), "UNKNOWN"
           )
           # Build a one-line summary
           concept = ex.get("concept_text", "")[:40]
           misconception = ex.get("misconception", "")
           one_liner = f"{concept} ({correct_label})"
           if misconception:
               one_liner += f" — {misconception[:50]}"

           compact_exchanges.append({
               "exchange_id": ex.get("exchange_id"),
               "concept": concept,
               "correct": ex.get("answer_correct"),
               "error_type": ex.get("error_type") or None,
               "one_liner": one_liner,
               "teaching_approach": ex.get("teaching_approach") or None,
               "teaching_worked": ex.get("teaching_worked"),
               "date": (ex.get("session_ts") or "")[:10],
               "skill": ex.get("skill"),
           })

       compact_summaries = []
       for s in full.get("episode_summaries", []):
           compact_summaries.append({
               "summary_id": s.get("summary_id"),
               "session_ts": (s.get("session_ts") or "")[:10],
               "exchange_count": s.get("exchange_count", 0),
               "correct_count": s.get("correct_count", 0),
               "incorrect_count": s.get("incorrect_count", 0),
               # Truncate memory_text to first 150 chars
               "memory_preview": (s.get("memory_text") or "")[:150],
           })

       return {
           "exchanges": compact_exchanges,
           "episode_summaries": compact_summaries,
           "patterns": full.get("patterns", {}),
           "full_ids_available": [e["exchange_id"] for e in compact_exchanges],
       }
   ```

PHASE 3: PRECOMPACT HOOK — RE-INJECT MEMORY BEFORE CONTEXT COMPACTION
================================================================================

STATUS: COMPLETE

GOAL: When Claude Code compresses the conversation context (compaction),
critical learning state is lost. We need a hook that fires before compaction
and re-injects the most important memory into the compressed context. This
ensures the agent remembers what happened earlier in long sessions.

BACKGROUND: Claude Code supports hooks — shell commands that execute in
response to lifecycle events. The relevant event is `PreCompact`, which fires
before context window compaction. The hook receives the conversation state on
stdin and can output text that gets injected into the compacted context.

IMPORTANT: Claude Code hooks are configured in the user's settings file at
~/.claude/settings.json (or the project-level .claude/settings.json). They
are NOT Python scripts in src/ — they are shell commands configured in JSON.

--- File: src/precompact_memory_inject.py (NEW FILE) ---

Create a new Python script that:
1. Reads the current session state (what skill is active, what topics)
2. Queries the knowledge graph for the current session's exchanges so far
3. Queries for the most relevant episodic memories for those topics
4. Outputs a compact memory block to stdout that Claude Code will inject

```python
#!/usr/bin/env python3
"""PreCompact memory injection — called by Claude Code hook before context compaction.

Reads stdin for the hook payload, queries the knowledge graph for current
session state and relevant episodic memories, outputs a compact memory
block that gets injected into the compacted context.

Usage (as Claude Code hook):
  python3 src/precompact_memory_inject.py
"""

from __future__ import annotations

import json
import sys
from datetime import datetime, timedelta, timezone
from pathlib import Path

BASE_DIR = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(BASE_DIR / "src"))


def main() -> int:
    # Read hook payload from stdin
    try:
        payload = json.load(sys.stdin)
    except (json.JSONDecodeError, Exception):
        payload = {}

    # Import KG
    try:
        from knowledge_graph import KnowledgeGraph
        kg = KnowledgeGraph()
    except Exception:
        return 0  # silent failure — don't block compaction

    try:
        now = datetime.now(timezone.utc)
        two_hours_ago = (now - timedelta(hours=2)).isoformat()

        # Find exchanges from the current session (last 2 hours)
        recent = kg.conn.execute(
            """SELECT le.concept_text, le.answer_correct, le.error_type,
                      le.misconception, le.teaching_approach, le.question_text,
                      le.correction_text, t.display_name AS topic_display
               FROM learning_exchanges le
               LEFT JOIN topics t ON le.topic_id = t.topic_id
               WHERE le.session_ts >= ?
               ORDER BY le.session_ts DESC, le.turn_number DESC
               LIMIT 15""",
            (two_hours_ago,),
        ).fetchall()

        if not recent:
            kg.close()
            return 0  # nothing to inject

        # Build compact memory block
        lines = ["## Active Session Memory (auto-injected before compaction)"]
        lines.append("")

        # Current session summary
        topics = sorted(set(
            r["topic_display"] or r["concept_text"] for r in recent
        ))
        correct = sum(1 for r in recent if r["answer_correct"] == 2)
        incorrect = sum(1 for r in recent if r["answer_correct"] == 0)
        partial = sum(1 for r in recent if r["answer_correct"] == 1)
        lines.append(f"Topics this session: {', '.join(topics[:5])}")
        lines.append(f"Exchanges: {len(recent)} total, {correct} correct, "
                      f"{partial} partial, {incorrect} incorrect")
        lines.append("")

        # Key errors to remember (most important for continuity)
        errors = [r for r in recent if r["answer_correct"] == 0]
        if errors:
            lines.append("Errors to re-test:")
            for e in errors[:5]:
                concept = e["concept_text"][:50]
                misconception = e["misconception"][:60] if e["misconception"] else ""
                q = e["question_text"][:60]
                correction = e["correction_text"][:60] if e["correction_text"] else ""
                line = f"  - {concept}"
                if misconception:
                    line += f" (misconception: {misconception})"
                if correction:
                    line += f" [correct: {correction}]"
                lines.append(line)
            lines.append("")

        # Teaching approach notes
        approaches_used = set()
        for r in recent:
            ta = r["teaching_approach"]
            if ta:
                approaches_used.add(ta)
        if approaches_used:
            lines.append(f"Teaching approaches used: {', '.join(sorted(approaches_used))}")

        # Relevant prior episodic memory (from before this session)
        if topics:
            topic_query = " ".join(topics[:3])
            try:
                prior = kg.recall_episodes_compact(
                    query=topic_query, max_results=3, days_back=30,
                )
                prior_exchanges = prior.get("exchanges", [])
                # Filter out current session exchanges
                prior_exchanges = [
                    e for e in prior_exchanges
                    if e.get("date", "") < two_hours_ago[:10]
                ]
                if prior_exchanges:
                    lines.append("")
                    lines.append("Relevant prior session memory:")
                    for pe in prior_exchanges[:3]:
                        lines.append(f"  - {pe.get('one_liner', 'unknown')}")
            except Exception:
                pass  # non-fatal

        lines.append("")
        lines.append("(End of injected session memory)")

        # Output to stdout — Claude Code will inject this
        output = "\n".join(lines)
        print(output)

        kg.close()
    except Exception:
        try:
            kg.close()
        except Exception:
            pass
        return 0  # never block compaction

    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

PHASE 4: MEMORY VERSIONING WITH SUPERSESSION CHAINS
================================================================================

STATUS: COMPLETE

GOAL: Track how the learner's understanding of concepts evolves over time.
Currently, concept_mastery rows are updated in place — when a concept
transitions from unknown to known, the old misconception is overwritten.
We lose the history of what the learner used to believe and how they
corrected it. Memory versioning creates an audit trail.

This is inspired by agentmemory's supersession chains (parentId +
supersedes[] on Memory objects) but adapted for our pedagogical context:
we care about CONCEPT UNDERSTANDING EVOLUTION, not generic memory updates.

WHAT TO CHANGE:

--- File: src/knowledge_graph.py ---

1. Add a new table to _SCHEMA_SQL (after the episode_summaries table,
   before the closing `"""`). This table tracks concept state changes:

   ```sql
   -- Concept evolution history — tracks how understanding changes over time.
   -- One row per state change on a concept_mastery row. Forms a temporal chain
   -- so the agent can see: "Gabriel first thought X, then corrected to Y,
   -- then regressed to Z, then finally stabilized on W."
   CREATE TABLE IF NOT EXISTS concept_evolution (
       evolution_id   INTEGER PRIMARY KEY AUTOINCREMENT,
       concept_id     INTEGER NOT NULL,
       topic_id       INTEGER NOT NULL,
       timestamp      TEXT NOT NULL,

       -- Snapshot of state AT THIS POINT
       status         TEXT NOT NULL,          -- 'known' | 'unknown'
       error_type     TEXT DEFAULT '',
       misconception  TEXT DEFAULT '',
       remediation    TEXT DEFAULT '',
       times_confirmed INTEGER DEFAULT 0,
       times_missed   INTEGER DEFAULT 0,

       -- What caused this change
       trigger_type   TEXT NOT NULL,          -- 'correct_recall' | 'incorrect_recall' | 'partial_recall' | 'teaching' | 'decay'
       trigger_exchange_id INTEGER,           -- FK to learning_exchanges (if change was triggered by an exchange)
       trigger_signal_id   INTEGER,           -- FK to signal_events (if triggered by a signal)

       -- Evolution chain
       previous_status TEXT DEFAULT '',       -- what status was BEFORE this change
       previous_misconception TEXT DEFAULT '', -- what misconception was BEFORE

       -- Natural language note
       evolution_note TEXT DEFAULT '',        -- e.g., "Corrected AChA -> PCA confusion via anatomical tracing"

       FOREIGN KEY (concept_id) REFERENCES concept_mastery(concept_id),
       FOREIGN KEY (topic_id) REFERENCES topics(topic_id),
       FOREIGN KEY (trigger_exchange_id) REFERENCES learning_exchanges(exchange_id),
       FOREIGN KEY (trigger_signal_id) REFERENCES signal_events(event_id)
   );
   CREATE INDEX IF NOT EXISTS idx_evo_concept ON concept_evolution(concept_id);
   CREATE INDEX IF NOT EXISTS idx_evo_topic ON concept_evolution(topic_id);
   CREATE INDEX IF NOT EXISTS idx_evo_ts ON concept_evolution(timestamp);
   ```

   ```python
       trigger_type: str,
       previous_status: str = "",
       previous_misconception: str = "",
       error_type: str = "",

   ) -> int:
       """Record a concept state change in the evolution history.

       Called automatically when concept_mastery is updated via

       """
       try:
           now = datetime.now(timezone.utc).isoformat()
           with self.conn:
               cur = self.conn.execute(
                   """INSERT INTO concept_evolution

                       error_type, misconception, remediation,
                       times_confirmed, times_missed,
                       trigger_type, trigger_exchange_id, trigger_signal_id,
                       previous_status, previous_misconception, evolution_note)
                      VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)""",
                   (concept_id, topic_id, now, new_status,
                    error_type, misconception, remediation,
                    times_confirmed, times_missed,
                    trigger_type, trigger_exchange_id, trigger_signal_id,
                    previous_status, previous_misconception, evolution_note),
               )

       except Exception as exc:

   ```

3. Add a method to query concept evolution history:

   ```python
   def concept_evolution_history(
       self,
       concept_text: str | None = None,
       topic_name: str | None = None,
       days_back: int = 180,
       limit: int = 50,
   ) -> list[dict]:
       """Query the evolution history for a concept or topic.

       Returns a list of state-change records ordered by timestamp DESC.
       """
       try:
           cutoff = (datetime.now(timezone.utc) - timedelta(days=days_back)).isoformat()
           clauses = ["ce.timestamp >= ?"]
           params: list = [cutoff]

           if concept_text:
               clauses.append("cm.concept_text LIKE ?")
               params.append(f"%{concept_text.strip().lower()}%")
           if topic_name:
               clauses.append("t.display_name LIKE ?")
               params.append(f"%{topic_name}%")

           where = " AND ".join(clauses)
           params.append(limit)

           rows = self.conn.execute(
               f"""SELECT ce.*, cm.concept_text, t.display_name AS topic_display
                   FROM concept_evolution ce
                   JOIN concept_mastery cm ON ce.concept_id = cm.concept_id
                   LEFT JOIN topics t ON ce.topic_id = t.topic_id
                   WHERE {where}
                   ORDER BY ce.timestamp DESC
                   LIMIT ?""",
               params,
           ).fetchall()
           return [dict(r) for r in rows]
       except Exception as exc:
           print(f"[knowledge_graph] concept_evolution_history error: {exc}", file=sys.stderr)
           return []
   ```
